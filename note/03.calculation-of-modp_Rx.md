# step by step of a calculation of [modp_Rx](https://github.com/open-quantum-safe/liboqs/blob/main/src/sig/falcon/pqclean_falcon-1024_clean/keygen.c#L758-L783)

```
/*
 * Compute 2^(31*x) modulo p. This works for integers x up to 2^11.
 * p must be prime such that 2^30 < p < 2^31; p0i must be equal to
 * -1/p mod 2^31; R2 must be equal to 2^62 mod p.
 */
static inline uint32_t
modp_Rx(unsigned x, uint32_t p, uint32_t p0i, uint32_t R2) {
    int i;
    uint32_t r, z;

    /*
     * 2^(31*x) = (2^31)*(2^(31*(x-1))); i.e. we want the Montgomery
     * representation of (2^31)^e mod p, where e = x-1.
     * R2 is 2^31 in Montgomery representation.
     */
    x --;
    r = R2;
    z = modp_R(p);
    for (i = 0; (1U << i) <= x; i ++) {
        if ((x & (1U << i)) != 0) {
            z = modp_montymul(z, r, p, p0i);
        }
        r = modp_montymul(r, r, p, p0i);
    }
    return z;
}
```

`z = modp_R(p)` is $$\text{"}2^{31} \mod{p}\text{"}$$, which is `1` in Montgomery representation.

(i.e. $$z = 1 \cdot c \mod{p}$$, where $$c=2^{31}$$ and $$2^{30} < p < 2^{31}$$)

`r = R2 = modp_R2()` is $$\text{"}2^{62} \mod{p}\text{"}$$, which is $$2^{31}$$ in Montgomery representation.,

(i.e. $$r = 2^{31} \cdot c  \mod{p}$$, where $$c=2^{31}$$ and $$2^{30} < p < 2^{31}$$)


- (note) the Montgomery Multiplication :

$$
\begin{aligned}
\overline{x} * \overline{y} = \overline{x} \cdot \overline{y} \cdot c^{-1} \mod{p} \\
\\
(\text{where } c \text{ and } p \text{ are integers such that } c > p \text{ and } c \text{ is coprime to } p),
\end{aligned}
$$

Let's compute when x = 10,

$$
\begin{aligned}
x &= 10_{10} \\
&= 1010_2
\end{aligned}
$$

$$
\begin{aligned}
2^{31 \cdot 10} &= 2^{31} \cdot 2^{31 \cdot 9} \\
\end{aligned}
$$

$$
\begin{aligned}
z_{base} &= R &= 2^{31}\mod{p} \\
r_{base} &= R2 &= 2^{62}\mod{p}
\end{aligned}
$$

$$
\begin{aligned}
x - 1 &=9_{10} \\
&= 1001_2
\end{aligned}
$$

i=0 :

$$
\begin{aligned}
x \And (1U << 0 ) &= 1001_2 \And 1_2 &= 1 \not = 0
\end{aligned}
$$

$$
\begin{aligned}
z_0 &= z_{base} * r_{base}\\
&= 2^{31} \cdot 2^{62} \cdot 2^{-31} \\
&= 2^{62}
\end{aligned}
$$

$$
\begin{aligned}
r_0 &= r_{base} * r_{base}\\
&= 2^{62} \cdot 2^{62} \cdot 2^{-31} \\
&= 2^{124} \cdot 2^{-31}\\
&= 2^{93}
\end{aligned}
$$

i=1 :

$$
\begin{aligned}
x \And (1U << 1 ) &= 1001_2 \And 10_2 &= 0
\end{aligned}
$$

$$
\begin{aligned}
z \text{ is not updated.} 
\end{aligned}
$$

$$
\begin{aligned}
r_1 &= r_{base} * r_{base}\\
&= r_0 \cdot r_0 \cdot 2^{-31} \\
&= 2^{93} \cdot 2^{93} \cdot 2^{-31} \\
&= 2^{186} \cdot 2^{-31} \\
&= 2^{155}
\end{aligned}
$$

i=2 :

$$
\begin{aligned}
x \And (1U << 2 ) &= 1001_2 \And 100_2 &= 0
\end{aligned}
$$

$$
\begin{aligned}
z \text{ is not updated.}
\end{aligned}
$$

$$
\begin{aligned}
r_2 &= r_1 * r_1\\
&= r_1 \cdot r_1 \cdot 2^{-31} \\
&= 2^{155} \cdot 2^{155} \cdot 2^{-31} \\
&= 2^{310} \cdot 2^{-31} \\
&= 2^{279}
\end{aligned}
$$

i=3 :

$$
\begin{aligned}
x \And (1U << 3 ) &= 1001_2 \And 1000_2 &= 1000_2 \not = 0
\end{aligned}
$$

$$
\begin{aligned}
z_3 &= z_0 * r_2\\
&= 2^{62} \cdot 2^{279} \cdot 2^{-31} \\
&= 2^{341} \cdot 2^{-31} \\
&= 2^{310}
\end{aligned}
$$

$$
\begin{aligned}
r_3 &= r_2 * r_2\\
&= r_2 \cdot r_2 \cdot 2^{-31} \\
&= 2^{279} \cdot 2^{279} \cdot 2^{-31} \\
&= 2^{558} \cdot 2^{-31} \\
&= 2^{527}
\end{aligned}
$$

i=4 :

$$
\begin{aligned}
\text{(1U << 4 ) } = 10000_2 = 16_{10} <= x \\
\text{Therefore, exit the loop and returns } z_3 ( =2^{310} ) \text{ which is } 2^{31 \cdot x} \text{.}
\end{aligned}
$$

It's pretty much similar to calculating pow(x, k).

```
int power(int x, int k)
{
    int result = 1;
    while (k > 0) {
        if (k & 1 == 1)
        {
            result = result * x;
        }
        x = x * x;
        k = k >> 1; // k=k/2;
    }
    return result;
}
```
